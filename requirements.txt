import random
import os
from datetime import datetime

class JogoPedraPapelTesoura:
    def __init__(self):
        self.opcoes = {
            '1': 'Pedra',
            '2': 'Papel', 
            '3': 'Tesoura'
        }
        self.emoji = {
            'Pedra': '✊',
            'Papel': '🖐️',
            'Tesoura': '✌️'
        }
        self.placar = {'jogador': 0, 'computador': 0, 'empates': 0}
        self.arquivo_placar = 'placar.txt'
        self.carregar_placar()
    
    def carregar_placar(self):
        """Carrega o placar do arquivo .txt"""
        if os.path.exists(self.arquivo_placar):
            try:
                with open(self.arquivo_placar, 'r', encoding='utf-8') as f:
                    linhas = f.readlines()
                    for linha in linhas:
                        if 'Jogador:' in linha:
                            self.placar['jogador'] = int(linha.split(':')[1].strip())
                        elif 'Computador:' in linha:
                            self.placar['computador'] = int(linha.split(':')[1].strip())
                        elif 'Empates:' in linha:
                            self.placar['empates'] = int(linha.split(':')[1].strip())
            except:
                print("Erro ao carregar placar. Iniciando com placar zerado.")
    
    def salvar_placar(self):
        """Salva o placar no arquivo .txt"""
        try:
            with open(self.arquivo_placar, 'w', encoding='utf-8') as f:
                f.write(f"=== PLACAR PEDRA, PAPEL E TESOURA ===\n")
                f.write(f"Data: {datetime.now().strftime('%d/%m/%Y %H:%M:%S')}\n")
                f.write(f"Jogador: {self.placar['jogador']}\n")
                f.write(f"Computador: {self.placar['computador']}\n")
                f.write(f"Empates: {self.placar['empates']}\n")
                f.write("=" * 35)
        except:
            print("Erro ao salvar placar.")
    
    def exibir_menu(self):
        """Exibe o menu principal do jogo"""
        print("\n" + "="*50)
        print("🎮 PEDRA, PAPEL E TESOURA 🎮")
        print("="*50)
        print("Escolha sua jogada:")
        print("1 - Pedra ✊")
        print("2 - Papel 🖐️") 
        print("3 - Tesoura ✌️")
        print("4 - Ver Placar")
        print("5 - Sair")
        print("="*50)
    
    def obter_jogada_jogador(self):
        """Obtém a jogada do jogador"""
        while True:
            try:
                escolha = input("\nDigite sua escolha (1-5): ").strip()
                if escolha in ['1', '2', '3']:
                    return self.opcoes[escolha]
                elif escolha == '4':
                    self.mostrar_placar()
                    continue
                elif escolha == '5':
                    return 'sair'
                else:
                    print("❌ Opção inválida! Escolha entre 1-5.")
            except KeyboardInterrupt:
                print("\n\n👋 Saindo do jogo...")
                return 'sair'
    
    def obter_jogada_computador(self):
        """Gera a jogada aleatória do computador"""
        return random.choice(list(self.opcoes.values()))
    
    def determinar_vencedor(self, jogada_jogador, jogada_computador):
        """Determina o vencedor da rodada"""
        if jogada_jogador == jogada_computador:
            return 'empate'
        
        vitorias = {
            'Pedra': 'Tesoura',
            'Papel': 'Pedra', 
            'Tesoura': 'Papel'
        }
        
        if vitorias[jogada_jogador] == jogada_computador:
            return 'jogador'
        else:
            return 'computador'
    
    def mostrar_resultado(self, jogada_jogador, jogada_computador, resultado):
        """Exibe o resultado da rodada"""
        print(f"\n{'='*30}")
        print(f"Você escolheu: {jogada_jogador} {self.emoji[jogada_jogador]}")
        print(f"Computador escolheu: {jogada_computador} {self.emoji[jogada_computador]}")
        print(f"{'='*30}")
        
        if resultado == 'empate':
            print("🤝 EMPATE!")
            self.placar['empates'] += 1
        elif resultado == 'jogador':
            print("🎉 VOCÊ VENCEU!")
            self.placar['jogador'] += 1
        else:
            print("😢 COMPUTADOR VENCEU!")
            self.placar['computador'] += 1
        
        print(f"{'='*30}")
    
    def mostrar_placar(self):
        """Exibe o placar atual"""
        print(f"\n{'='*25}")
        print("📊 PLACAR ATUAL")
        print(f"{'='*25}")
        print(f"Jogador: {self.placar['jogador']} vitórias")
        print(f"Computador: {self.placar['computador']} vitórias") 
        print(f"Empates: {self.placar['empates']}")
        print(f"{'='*25}")
    
    def jogar(self):
        """Loop principal do jogo"""
        print("🎮 Bem-vindo ao Pedra, Papel e Tesoura!")
        
        while True:
            self.exibir_menu()
            jogada_jogador = self.obter_jogada_jogador()
            
            if jogada_jogador == 'sair':
                self.salvar_placar()
                print("\n👋 Obrigado por jogar! Placar salvo.")
                break
            
            jogada_computador = self.obter_jogada_computador()
            resultado = self.determinar_vencedor(jogada_jogador, jogada_computador)
            
            self.mostrar_resultado(jogada_jogador, jogada_computador, resultado)
            self.salvar_placar()
            
            input("\nPressione Enter para continuar...")

def main():
    """Função principal"""
    jogo = JogoPedraPapelTesoura()
    jogo.jogar()

if __name__ == "__main__":
    main()
